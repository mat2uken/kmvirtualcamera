cmake_minimum_required(VERSION 3.20)
project(KMVirtualCameraFoundation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(KM_BUILD_TESTS "Build dependency-free common tests" ON)
option(KM_BUILD_MACOS "Build unsigned native media/relay smoke targets (not a distributable app)" OFF)
option(KM_ENABLE_SANITIZERS "Address/undefined sanitizers on supported compilers" OFF)
add_library(km_media_core STATIC shared/receiver/codec/dc_video_depacketizer.cpp)
target_include_directories(km_media_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/shared)
if(MSVC)
    target_compile_options(km_media_core PRIVATE /W4 /utf-8)
else()
    target_compile_options(km_media_core PRIVATE -Wall -Wextra -Wpedantic)
    if(KM_ENABLE_SANITIZERS)
        target_compile_options(km_media_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(km_media_core PUBLIC -fsanitize=address,undefined)
    endif()
endif()
if(KM_BUILD_TESTS)
    enable_testing()
    add_executable(km_core_tests tests/shared/test_core.cpp)
    target_link_libraries(km_core_tests PRIVATE km_media_core)
    add_test(NAME common_contracts COMMAND km_core_tests)
    add_executable(km_dc_tests tests/shared/test_dc.cpp)
    target_link_libraries(km_dc_tests PRIVATE km_media_core)
    add_test(NAME datachannel_regression COMMAND km_dc_tests)
endif()
if(KM_BUILD_MACOS)
    if(NOT APPLE)
        message(FATAL_ERROR "KM_BUILD_MACOS requires macOS and Xcode SDK")
    endif()
    enable_language(OBJCXX)
    add_subdirectory(macos)
endif()
# The existing production Windows build remains: cmake -S windows -B build/windows.
# Do not link km_media_core into Receiver while its transitional forwarding TU is compiled.

option(KM_BUILD_RTC_SHARED "Build extracted legacy RTP/RTCP modules using an installed libdatachannel" OFF)
if(KM_BUILD_RTC_SHARED)
    find_package(LibDataChannel CONFIG REQUIRED)
    add_library(km_rtc_shared STATIC
        shared/receiver/codec/h264_rtp_depacketizer.cpp
        shared/receiver/rtc/bandwidth_estimator.cpp
        shared/receiver/rtc/twcc_receiver.cpp
        shared/receiver/rtc/enhanced_rtcp_session.cpp)
    target_include_directories(km_rtc_shared PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/shared)
    target_link_libraries(km_rtc_shared PUBLIC km_media_core LibDataChannel::LibDataChannel)
endif()
